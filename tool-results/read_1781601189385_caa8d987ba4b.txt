    34→export default function Home() {
    35→  const { isAuthenticated, user, token, login, logout } = useAuthStore();
    36→  const [view, setView] = useState<View>('landing');
    37→  const [showAuthDialog, setShowAuthDialog] = useState(false);
    38→  const [authMode, setAuthMode] = useState<'login' | 'register'>('login');
    39→  const [loading, setLoading] = useState(false);
    40→  const [authKey, setAuthKey] = useState(0);
    41→
    42→  // Check if user is authenticated on mount (use getState for reliable hydration)
    43→  useEffect(() => {
    44→    const { token: storedToken, isAuthenticated: isAuth } = useAuthStore.getState();
    45→    if (storedToken && isAuth) {
    46→      setView('dashboard');
    47→      // Verify token is still valid — only logout on auth errors, not network errors
    48→      api.getMe().catch((err: any) => {
    49→        const msg = err?.message || '';
    50→        if (msg.includes('Unauthorized') || msg.includes('401') || msg.includes('Token')) {
    51→          logout();
    52→          setView('landing');
    53→        }
    54→      });
    55→    }
    56→  }, []);
    57→
    58→  const handleLogin = useCallback(async (identifier: string, password: string) => {
    59→    setLoading(true);
    60→    try {
    61→      const result = await api.login(identifier, password);
    62→      login(result.token, result.user);
    63→      setShowAuthDialog(false);
    64→      setView('dashboard');
    65→    } catch (err: any) {
    66→      throw new Error(err.message || 'Login failed');
    67→    } finally {
    68→      setLoading(false);
    69→    }
    70→  }, [login]);
    71→
    72→  const handleRegister = useCallback(async (data: any) => {
    73→    setLoading(true);
    74→    try {
    75→      const result = await api.register(data);
    76→      login(result.token, result.user);
    77→      setShowAuthDialog(false);
    78→      setView('dashboard');
    79→    } catch (err: any) {
    80→      throw new Error(err.message || 'Registration failed');
    81→    } finally {
    82→      setLoading(false);
    83→    }
    84→  }, [login]);
    85→
    86→  const handleLogout = useCallback(() => {
    87→    logout();
    88→    setView('landing');
    89→  }, [logout]);
    90→
    91→  const openAuth = useCallback((mode: 'login' | 'register') => {
    92→    setAuthMode(mode);
    93→    setAuthKey((k) => k + 1);
    94→    setShowAuthDialog(true);
    95→  }, []);
    96→
    97→  if (view === 'dashboard' && isAuthenticated && user) {
    98→    return (
    99→      <DashboardView
   100→        user={user}
   101→        onLogout={handleLogout}
   102→      />
   103→    );
   104→  }
   105→
   106→  return (
   107→    <LandingView
   108→      onOpenAuth={openAuth}
   109→      showAuthDialog={showAuthDialog}
   110→      authMode={authMode}
   111→      authKey={authKey}
   112→      onCloseAuth={() => setShowAuthDialog(false)}
   113→      onSwitchAuthMode={() => setAuthMode(authMode === 'login' ? 'register' : 'login')}
   114→      onLogin={handleLogin}
   115→      onRegister={handleRegister}
   116→      loading={loading}
   117→    />
   118→  );
   119→}

   384→// ==================== AUTH DIALOG ====================
   385→function AuthDialog({
   386→  mode,
   387→  onClose,
   388→  onLogin,
   389→  onRegister,
   390→  onSwitchMode,
   391→  loading,
   392→}: {
   393→  mode: 'login' | 'register';
   394→  onClose: () => void;
   395→  onLogin: (identifier: string, password: string) => Promise<void>;
   396→  onRegister: (data: any) => Promise<void>;
   397→  onSwitchMode: () => void;
   398→  loading: boolean;
   399→}) {
   400→  // Login state
   401→  const [loginMethod, setLoginMethod] = useState<'email' | 'username' | 'phone'>('email');
   402→  const [loginIdentifier, setLoginIdentifier] = useState('');
   403→  // Shared / register state
   404→  const [email, setEmail] = useState('');
   405→  const [password, setPassword] = useState('');
   406→  const [showPassword, setShowPassword] = useState(false);
   407→  const [name, setName] = useState('');
   408→  const [firstName, setFirstName] = useState('');
   409→  const [middleName, setMiddleName] = useState('');
   410→  const [lastName, setLastName] = useState('');
   411→  const [username, setUsername] = useState('');
   412→  const [phone, setPhone] = useState('');
   413→  const [role, setRole] = useState<'CLIENT' | 'PROVIDER'>('CLIENT');
   414→  const [selectedSkills, setSelectedSkills] = useState<string[]>([]);
   415→  const [skillInput, setSkillInput] = useState('');
   416→  const [hourlyRate, setHourlyRate] = useState('');
   417→  const [location, setLocation] = useState('');
   418→  const [bio, setBio] = useState('');
   419→  const [bankName, setBankName] = useState('');
   420→  const [accountNumber, setAccountNumber] = useState('');
   421→  const [accountName, setAccountName] = useState('');
   422→  const [error, setError] = useState('');
   423→  const [confirmPassword, setConfirmPassword] = useState('');
   424→
   425→  // Password strength calculator
   426→  const getPasswordStrength = (pw: string) => {
   427→    if (!pw) return { label: '', color: '', width: '0%' };
   428→    let score = 0;
   429→    if (pw.length >= 6) score++;
   430→    if (pw.length >= 8) score++;
   431→    if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
   432→    if (/\d/.test(pw)) score++;
   433→    if (/[^a-zA-Z0-9]/.test(pw)) score++;
   434→    if (pw.length >= 12) score++;
   435→    if (score <= 2) return { label: 'Weak', color: 'bg-red-500', width: '25%' };
   436→    if (score <= 3) return { label: 'Fair', color: 'bg-orange-500', width: '50%' };
   437→    if (score <= 4) return { label: 'Good', color: 'bg-yellow-500', width: '75%' };
   438→    return { label: 'Strong', color: 'bg-green-500', width: '100%' };
   439→  };
   440→  const pwStrength = mode === 'register' ? getPasswordStrength(password) : null;
   441→
   442→  const addSkill = (value: string) => {
   443→    const trimmed = value.trim();
   444→    if (trimmed && !selectedSkills.some(s => s.toLowerCase() === trimmed.toLowerCase())) {
   445→      setSelectedSkills(prev => [...prev, trimmed]);
   446→    }
   447→    setSkillInput('');
   448→  };
   449→
   450→  const removeSkill = (skill: string) => {
   451→    setSelectedSkills(prev => prev.filter(s => s !== skill));
   452→  };
   453→
   454→  const handleSkillKeyDown = (e: React.KeyboardEvent) => {
   455→    if ((e.key === 'Enter' || e.key === ',') && skillInput.trim()) {
   456→      e.preventDefault();
   457→      addSkill(skillInput);
   458→    } else if (e.key === 'Backspace' && !skillInput && selectedSkills.length > 0) {
   459→      removeSkill(selectedSkills[selectedSkills.length - 1]);
   460→    }
   461→  };
   462→
   463→  const handleSubmit = async (e: React.FormEvent) => {
   464→    e.preventDefault();
   465→    setError('');
   466→    if (mode === 'register' && role === 'PROVIDER' && selectedSkills.length === 0) {
   467→      setError('Please add at least one service you offer');
   468→      return;
   469→    }
   470→    if (mode === 'register' && username.trim().length < 3) {
   471→      setError('Username is required (minimum 3 characters)');
   472→      return;
   473→    }
   474→    if (mode === 'register' && password !== confirmPassword) {
   475→      setError('Passwords do not match');
   476→      return;

   888→function ProviderRestrictedView({ user, verificationStatus, onLogout }: { user: any; verificationStatus: string; onLogout: () => void }) {

   910→          if (me.provider?.verificationStatus === 'VERIFIED') {
   911→            useAuthStore.getState().login(authToken, me);

   913→        }
   914→      } catch {}
   915→    };

   921→  const headers = (): Record<string, string> => {
   922→    const h: Record<string, string> = {};
   923→    if (authToken) h['Authorization'] = `Bearer ${authToken}`;
   924→    return h;

   933→        const errData = await res.json().catch(() => ({}));
   934→        throw new Error(errData.detail || errData.error || `HTTP ${res.status}`);
   935→      }
   936→      const data = await res.json();
   937→      if (data.messages) setMessages(data.messages);
   938→    } catch (err: any) {
   939→      console.error('[SupportChat] Fetch error:', err);
   940→      setChatError(err.message || 'Failed to load messages');
   941→    }
   942→  }, [authToken]);
   943→
   944→  useEffect(() => {
   945→    setChatLoading(true);
   946→    fetchMessages().finally(() => setChatLoading(false));
   947→    const interval = setInterval(fetchMessages, 5000);
   948→    return () => clearInterval(interval);
   949→  }, [fetchMessages]);
   950→
   951→  useEffect(() => {
   952→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   953→  }, [messages]);
   954→
   955→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   956→    const file = e.target.files?.[0];

   958→    setUploading(true);
   959→    setChatError('');
   960→    try {
   961→      const formData = new FormData();
